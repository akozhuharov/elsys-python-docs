# Направте клас Calculator с 4 функции - събиране, изваждане, умножение и деление + конструктор с 2 параметъра
```python
class Calculator():
    def __init__(self, a, b):
        self.a = a
        self.b = b

    def add(self):
        return self.a + self.b

    def substr(self):
        return self.a - self.b

    def divide(self):
        return self.a / self.b

    def mult(self):
        return self.a * self.b


c = Calculator(15, 4)
print(c.add())
print(c.divide())
print(c.mult())
print(c.substr())
```
